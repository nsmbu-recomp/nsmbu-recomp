cmake_minimum_required(VERSION 3.20)
project(nsmbu_recomp C CXX)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(NSMBU_IOS ON)
else()
  set(NSMBU_IOS OFF)
endif()

set(NSMBU_RENDERER "" CACHE STRING "Graphics backends: BOTH, METAL or VULKAN")
if(NOT NSMBU_RENDERER)
  if(NSMBU_IOS)
    set(NSMBU_RENDERER METAL)
  elseif(APPLE)
    set(NSMBU_RENDERER BOTH)
  else()
    set(NSMBU_RENDERER VULKAN)
  endif()
endif()
string(TOUPPER "${NSMBU_RENDERER}" NSMBU_RENDERER)
set_property(CACHE NSMBU_RENDERER PROPERTY STRINGS BOTH METAL VULKAN)
if(NOT NSMBU_RENDERER MATCHES "^(BOTH|METAL|VULKAN)$")
  message(FATAL_ERROR "NSMBU_RENDERER must be BOTH, METAL or VULKAN")
endif()
set(NSMBU_HAS_METAL OFF)
set(NSMBU_HAS_VULKAN OFF)
if(NSMBU_RENDERER MATCHES "^(BOTH|METAL)$")
  if(NOT APPLE)
    message(FATAL_ERROR "Metal is available only on macOS; select VULKAN")
  endif()
  set(NSMBU_HAS_METAL ON)
  enable_language(OBJCXX)
endif()
if(NSMBU_RENDERER MATCHES "^(BOTH|VULKAN)$")
  if(NSMBU_IOS)
    message(FATAL_ERROR "the iOS build uses the Metal renderer only; select METAL")
  endif()
  set(NSMBU_HAS_VULKAN ON)
endif()

if(NSMBU_HAS_VULKAN AND NOT NSMBU_HAS_METAL)
  set(NSMBU_SDL_HOST ON)
else()
  set(NSMBU_SDL_HOST OFF)
endif()
message(STATUS "NSMBU renderers: ${NSMBU_RENDERER} (Metal ${NSMBU_HAS_METAL}, Vulkan ${NSMBU_HAS_VULKAN}, SDL host ${NSMBU_SDL_HOST})")
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "The recompiled game requires Clang (musttail support); select clang/clang++")
endif()

if(CMAKE_SYSTEM_NAME MATCHES "^(Linux|Android)$" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  add_compile_options(-fsigned-char)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

set(GEN_DIR ${CMAKE_SOURCE_DIR}/build/gen CACHE PATH "recompiler output")
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS ${GEN_DIR}/code_*.c)
if(NOT GEN_SOURCES)
  message(FATAL_ERROR "no generated code in ${GEN_DIR}; run tools/recomp/recomp.py first")
endif()

option(NSMBU_BUNDLED_DEPS "build missing dependencies from pinned sources" OFF)

  include(cmake/WindowsDependencies.cmake)
endif()
if(WIN32 AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()

if(NSMBU_IOS)
  target_compile_options(gamecode PRIVATE -mcpu=${NSMBU_IOS_CPU})
elseif(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  target_compile_options(gamecode PRIVATE -mcpu=apple-m1)
endif()
